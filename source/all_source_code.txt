void Cmd_Cmatrix() {
	std::srand(static_cast<unsigned int>(std::time(nullptr)));
	system("cls");
	system("color 2");

	std::string chars = "QWERTYUIOPLKJHGFDSA2346";
	int width = 160;

	
	std::vector<int> rain(width, 0);

	while (true) {
		
		if (GetAsyncKeyState(VK_ESCAPE) & 0x8000) {
			std::cout << "\nback..." << std::endl;
			system("cls");
			system("color 07");

			Sleep(500);
			return;
		}

		
		for (int i = 0; i < width; i++) {
			if (rain[i] > 0) {
				
				std::cout << chars[std::rand() % chars.length()];
				rain[i]--;
			}
			else {
				
				std::cout << ' ';

				
				if (std::rand() % 10 == 0) {
					rain[i] = 10 + std::rand() % 30;
				}
			}
		}
		std::cout << '\n';

		
		std::this_thread::sleep_for(std::chrono::milliseconds(30));
	}
}

std::string getRegistryString(HKEY hKeyParent, const char* subKey, const char* valueName) {
    HKEY hKey;
    char value[256];
    DWORD value_length = sizeof(value);
    if (RegOpenKeyExA(hKeyParent, subKey, 0, KEY_READ, &hKey) == ERROR_SUCCESS) {
        if (RegQueryValueExA(hKey, valueName, NULL, NULL, (LPBYTE)value, &value_length) == ERROR_SUCCESS) {
            RegCloseKey(hKey);
            return std::string(value);
        }
        RegCloseKey(hKey);
    }
    return "Unknown";
}

void cmd_fastfetch() {
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);

    char username[UNLEN + 1];
    DWORD username_len = UNLEN + 1;
    GetUserNameA(username, &username_len);

    char computername[MAX_COMPUTERNAME_LENGTH + 1];
    DWORD computername_len = MAX_COMPUTERNAME_LENGTH + 1;
    GetComputerNameA(computername, &computername_len);

    SYSTEM_INFO sysInfo;
    GetSystemInfo(&sysInfo);
    std::string arch = "Unknown";
    if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_AMD64) arch = "x86_64";
    else if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_INTEL) arch = "x86";
    else if (sysInfo.wProcessorArchitecture == PROCESSOR_ARCHITECTURE_ARM64) arch = "ARM64";

    ULONGLONG uptimeMs = GetTickCount64();
    int uptimeHours = (int)(uptimeMs / (1000 * 60 * 60));
    int uptimeMinutes = (int)((uptimeMs / (1000 * 60)) % 60);

    MEMORYSTATUSEX memInfo;
    memInfo.dwLength = sizeof(MEMORYSTATUSEX);
    GlobalMemoryStatusEx(&memInfo);
    DWORDLONG totalRAM = memInfo.ullTotalPhys / (1024 * 1024);
    DWORDLONG freeRAM = memInfo.ullAvailPhys / (1024 * 1024);
    DWORDLONG usedRAM = totalRAM - freeRAM;
    int memPercent = (int)((usedRAM * 100) / totalRAM);

    std::string cpuName = getRegistryString(HKEY_LOCAL_MACHINE, "HARDWARE\\DESCRIPTION\\System\\CentralProcessor\\0", "ProcessorNameString");
    size_t first = cpuName.find_first_not_of(' ');
    if (std::string::npos != first) cpuName = cpuName.substr(first);

    std::string gpuName = getRegistryString(HKEY_LOCAL_MACHINE, "SOFTWARE\\Microsoft\\Windows NT\\CurrentVersion\\WinSAT", "PrimaryAdapterString");

        gpuName = getRegistryString(HKEY_LOCAL_MACHINE, "SYSTEM\\CurrentControlSet\\Control\\Class\\{4d36e968-e325-11ce-bfc1-08002be10318}\\0000", "DriverDesc");
    }

    int screenWidth = GetSystemMetrics(SM_CXSCREEN);
    int screenHeight = GetSystemMetrics(SM_CYSCREEN);

    auto setBlue = [&]() { SetConsoleTextAttribute(hConsole, 9); };
    auto setWhite = [&]() { SetConsoleTextAttribute(hConsole, 7); };

    std::cout << "\n";

    
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   "; setBlue();  std::cout << username; setWhite(); std::cout << "@"; setBlue(); std::cout << computername << "\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   -------------------\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   OS: Windows 11 / 10 " << arch << "\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   Kernel: Windows_NT\n";

    
    std::cout << "                   " << "   "; std::cout << " Uptime: " << uptimeHours << " hours, " << uptimeMinutes << " mins\n";

    
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   Shell: WindowsShell (C++)\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   Display: " << screenWidth << "x" << screenHeight << "\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   CPU: " << cpuName << "\n";
    setBlue();  std::cout << "█████████  █████████"; setWhite(); std::cout << "   GPU: " << gpuName << "\n";

    
    std::cout << "                   " << "   "; std::cout << " Memory: " << usedRAM << " MB / " << totalRAM << " MB (" << memPercent << "%)\n";
    std::cout << "                   " << "   "; std::cout << "\n";

void dev_collect_all_code() {
    std::string out_name = "all_source_code.txt";
    std::ofstream out(out_name);
    if (!out) return;
    try {
        for (const auto& entry : std::filesystem::directory_iterator(std::filesystem::current_path())) {
            if (entry.is_regular_file()) {
                std::string ext = entry.path().extension().string();
                if (ext == ".cpp" || ext == ".h" || ext == ".rc") {
                    std::string filename = entry.path().filename().string();
                    if (filename == out_name) continue;
                    std::ifstream in(entry.path());
                    if (in) {
                        out << "\n// ==================================================\n";
                        out << "// file: " << filename << "\n";
                        out << "// ==================================================\n\n";
                        std::string line;
                        while (std::getline(in, line)) {
                            out << line << "\n";
                        }
                        in.close();
                    }
                }
            }
        }
        out.close();
    }
    catch (...) {
        std::cerr << "dev_collect_all_code: failed\n";
    }
}

void cmd_touch(const std::string& filename) {
    if (filename.empty()) {
        std::cout << "Error: No filename specified.\n";
        return;
    }
    fs::path filePath(filename);
    try {
        if (!fs::exists(filePath)) {
            std::ofstream file(filePath);
            if (!file) {
                std::cerr << "Failed to create file: " << filename << "\n";
                return;
            }
            std::cout << "File created: " << filename << "\n";
        }
        else {
            auto currentTime = fs::file_time_type::clock::now();
            fs::last_write_time(filePath, currentTime);
            std::cout << "File modification time updated: " << filename << "\n";
        }
    }
    catch (const fs::filesystem_error& e) {
        std::cerr << "Filesystem error: " << e.what() << "\n";
    }
}

void cmd_ls() {
    try {
        fs::path currentDir = fs::current_path();
        if (fs::is_empty(currentDir)) {
            std::cout << "(directory is empty)\n";
            return;
        }
        HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);
        for (const auto& entry : fs::directory_iterator(currentDir)) {
            std::string name = entry.path().filename().string();
            if (entry.is_directory()) {
                SetConsoleTextAttribute(hConsole, 9);
                std::cout << name << "/\n";
                SetConsoleTextAttribute(hConsole, 7);
            }
            else {
                std::cout << name << "\n";
            }
        }
    }
    catch (const fs::filesystem_error& e) {
        std::cerr << "Error reading directory: " << e.what() << "\n";
    }
}

void cmd_cd(std::string args) {
    if (args.empty()) {
        std::cout << "Eror Try 'cd [directory]'\n";
        return;
    }
    try {
        fs::path target(args);
        if (args.size() == 2 && args[1] == ':') {
            target = args + "\\";
        }
        if (!fs::exists(target) || !fs::is_directory(target)) {
            std::cout << "Eror Directory not found: " << args << "\n";
            return;
        }
        fs::current_path(target);
    }
    catch (const fs::filesystem_error& e) {
        std::cerr << "Filesystem error: " << e.what() << "\n";
    }
}

void cmd_neo(std::string filename) {
    if (filename.empty()) {
        std::cout << "Error: No filename specified. Usage: neo [filename]\n";
        return;
    }

    std::filesystem::path filePath(filename);
    std::string text = "";

    if (std::filesystem::exists(filePath)) {
        std::ifstream inFile(filePath);
        if (inFile) {
            std::string line;
            while (std::getline(inFile, line)) {
                text = text + line + "\n";
            }
        }
        inFile.close();
    }

    system("cls");
    std::cout << "=== NEO Editor (File: " << filename << ") ===\n";
    std::cout << "[ Hotkeys: Ctrl+S - Save | Esc - Exit ]\n";
    std::cout << "--------------------------------------------------\n";
    std::cout << text;

    while (true) {
        int ch = _getch();

        if (ch == 19) {
            std::ofstream outFile(filePath);
            if (outFile) {
                outFile << text;
                outFile.close();
                std::cout << "\n[Saved!]";
                std::this_thread::sleep_for(std::chrono::milliseconds(600));
            }
            system("cls");
            std::cout << "=== NEO Editor (File: " << filename << ") ===\n";
            std::cout << "[ Hotkeys: Ctrl+S - Save | Esc - Exit ]\n";
            std::cout << "--------------------------------------------------\n";
            std::cout << text;
        }
        else if (ch == 27) {
            break;
        }
        else if (ch == 13) {
            text = text + "\n";
            std::cout << "\n";
        }
        else if (ch == 8) {
            if (text.empty() == false) {
                text.pop_back();
                system("cls");
                std::cout << "=== NEO Editor (File: " << filename << ") ===\n";
                std::cout << "[ Hotkeys: Ctrl+S - Save | Esc - Exit ]\n";
                std::cout << "--------------------------------------------------\n";
                std::cout << text;
            }
        }
        else if (ch >= 32) {
            text = text + (char)ch;
            std::cout << (char)ch;
        }
    }

    system("cls");
}

#include "pipes.h"
#include "includes.h"

struct Pipe {
    int x, y;
    int dir;
    int color;
};

void cmd_pipes() {
    HANDLE hConsole = GetStdHandle(STD_OUTPUT_HANDLE);
    srand(static_cast<unsigned int>(time(nullptr)));

    const char* glyphs[4][4] = {
        { "║", "╔", "║", "╗" },
        { "╚", "═", "╔", "═" },
        { "║", "╚", "║", "╝" },
        { "╝", "═", "╗", "═" }
    };

    int screenWidth = 80;
    int screenHeight = 25;
    CONSOLE_SCREEN_BUFFER_INFO csbi;
    if (GetConsoleScreenBufferInfo(hConsole, &csbi)) {
        screenWidth = csbi.srWindow.Right - csbi.srWindow.Left + 1;
        screenHeight = csbi.srWindow.Bottom - csbi.srWindow.Top + 1;
    }

    CONSOLE_CURSOR_INFO cursorInfo;
    GetConsoleCursorInfo(hConsole, &cursorInfo);
    cursorInfo.bVisible = FALSE;
    SetConsoleCursorInfo(hConsole, &cursorInfo);

    system("cls");

    const int maxPipes = 15;
    Pipe pipes[maxPipes];

    auto resetPipe = [&](Pipe& p) {
        p.x = rand() % screenWidth;
        p.y = rand() % screenHeight;
        p.dir = rand() % 4;
        p.color = 1 + (rand() % 14);
        if (p.color == 7 || p.color == 8) p.color = 9;
        };

    for (int i = 0; i < maxPipes; i++) {
        resetPipe(pipes[i]);
    }

    while (!_kbhit()) {
        for (int i = 0; i < maxPipes; i++) {
            int oldDir = pipes[i].dir;

            if (rand() % 4 == 0) {
                int turn = (rand() % 2 == 0) ? 1 : 3;
                pipes[i].dir = (pipes[i].dir + turn) % 4;
            }

            COORD pos = { static_cast<SHORT>(pipes[i].x), static_cast<SHORT>(pipes[i].y) };
            SetConsoleCursorPosition(hConsole, pos);
            SetConsoleTextAttribute(hConsole, pipes[i].color);
            std::cout << glyphs[oldDir][pipes[i].dir];

            if (pipes[i].dir == 0) pipes[i].y--;
            else if (pipes[i].dir == 1) pipes[i].x++;
            else if (pipes[i].dir == 2) pipes[i].y++;
            else if (pipes[i].dir == 3) pipes[i].x--;

            if (pipes[i].x < 0 || pipes[i].x >= screenWidth || pipes[i].y < 0 || pipes[i].y >= screenHeight) {
                resetPipe(pipes[i]);
            }
        }
        std::cout.flush();
        Sleep(60);
    }

    static_cast<void>(_getch());

    cursorInfo.bVisible = TRUE;
    SetConsoleCursorInfo(hConsole, &cursorInfo);
    SetConsoleTextAttribute(hConsole, 7);
    system("cls");
}

#include "PowerContor.h"
#include "includes.h"

void ShutDownPc() {
    std::cout << "shutdown pc in 5 seconds!" << std::endl;
    system("shutdown /s /f /t 5");
}

void RestartPc() {
    std::cout << "reboot pc in 5 seconds!" << std::endl;
    system("shutdown /r /f /t 5");
}

void SleepPc() {
    std::cout << "sleeping pc in 5 seconds!" << std::endl;
    std::this_thread::sleep_for(std::chrono::seconds(5));
    system("powercfg -h off");
    system("rundll32.exe powrprof.dll,SetSuspendState 0,1,0");
    system("powercfg -h on");
}

void SetsColor(std::string args) {
    if (args.empty()) {
        std::cout << "Eror Try 'color [number]'" << std::endl;
    }



    std::string command = "color " + args;

    system(command.c_str());
}

void Cmd_Beep(std::string arguments) {



    const char* spacePtr = strchr(arguments.c_str(), ' ');


    if (spacePtr == nullptr) {
        std::cout << "USE 'beep [power] [timer]'\n";
        return;
    }

    int power = atoi(arguments.c_str());
    int timer = atoi(strchr(arguments.c_str(), ' '));

    Beep(power * 100, timer * 1000);
}

void Cmd_Ping(std::string args) {
    if (args.empty()) {
        std::cout << "Error try 'ping example.com'" << std::endl;
    }

    std::string command = "ping " + args;

    system(command.c_str());
}

void Cmd_Echo(std::string args) {
    if (args.empty()) {
        std::cout << "Eror Try echo 'text'" << std::endl;
        return;
    }

    std::cout  << args << "\n";

}
